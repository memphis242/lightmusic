# FTXUI probe. Uses system FTXUI (Fedora: ftxui-devel) via find_package; if not found and
# LM_FETCH_FTXUI=ON, fetches FTXUI v5.0.0 with FetchContent (needs network) and links statically.
cmake_minimum_required(VERSION 3.14)
project(lm_ftxui_probe CXX)
set(CMAKE_CXX_STANDARD 17)
option(LM_FETCH_FTXUI "Fetch FTXUI if no system package" OFF)
find_package(ftxui CONFIG QUIET)
if(ftxui_FOUND)
  set(LM_FTXUI_SOURCE "system ${ftxui_VERSION}")
elseif(LM_FETCH_FTXUI)
  include(FetchContent)
  set(FTXUI_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(FTXUI_BUILD_DOCS OFF CACHE BOOL "" FORCE)
  set(FTXUI_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(ftxui GIT_REPOSITORY https://github.com/ArthurSonzogni/FTXUI GIT_TAG v5.0.0 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(ftxui)
  set(LM_FTXUI_SOURCE "fetched v5.0.0 (static)")
else()
  message(FATAL_ERROR "FTXUI not found (install ftxui-devel or pass -DLM_FETCH_FTXUI=ON)")
endif()
message(STATUS "LM_FTXUI_SOURCE=${LM_FTXUI_SOURCE}")
file(WRITE ${CMAKE_BINARY_DIR}/ftxui_source.txt "${LM_FTXUI_SOURCE}\n")
add_executable(ftxui_probe ${CMAKE_CURRENT_SOURCE_DIR}/../ftxui.cpp)
target_link_libraries(ftxui_probe PRIVATE ftxui::screen ftxui::dom ftxui::component)
